# Run from the DLKcat submodule root. Python: conda env mldb (PY=python after `conda activate mldb`).
# Default paths: data in EMULaToR/data/processed/datasets/kinetic_dataset/kcat,
# caches in EMULaToR/outputs/baselines/DLKcat/embeddings,
# results in EMULaToR/outputs/baselines/DLKcat/kcat/<split_group>/<split_name>/seed_<seed>.

# Build the feature cache (the launcher also does this unless --skip_cache is set).
$PY emulator_bench/cache_embeddings.py

# Train all discovered splits x seeds 666 777 888 on GPUs 0-3, with original settings.
$PY emulator_bench/launch_parallel_retrain.py \
  --gpus 0 1 2 3 \
  --runs_per_gpu 1

# All 21 runs at once on one GPU. Start CUDA MPS first so the processes overlap on the GPU (about 2.5x faster than time slicing):
export CUDA_MPS_PIPE_DIRECTORY=$HOME/mps CUDA_MPS_LOG_DIRECTORY=$HOME/mps/log
mkdir -p $CUDA_MPS_PIPE_DIRECTORY $CUDA_MPS_LOG_DIRECTORY
nvidia-cuda-mps-control -d
OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 $PY emulator_bench/launch_parallel_retrain.py --gpus 0 --runs_per_gpu 21
# Stop MPS afterwards: echo quit | nvidia-cuda-mps-control

# Train one split and one seed.
$PY emulator_bench/train_single_target_tvt.py \
  --split_group random_splits_grouped_sequence \
  --seed 666 \
  --device cuda:0

# Aggregate the outputs.
$PY emulator_bench/aggregate_dlkcat_results.py

# Equivalence check against the original code (CPU). Add --gpu for the CUDA test or --negative-control.
bash emulator_bench/verify_equivalence.sh
